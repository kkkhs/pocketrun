cmake_minimum_required(VERSION 3.22.1)
project(pocketrun_adb)
set(CMAKE_CXX_STANDARD 17)
find_package(boringssl REQUIRED CONFIG)
find_package(cxx REQUIRED CONFIG)
add_library(pocketrun_adb SHARED adb_pairing.cpp)
target_compile_options(pocketrun_adb PRIVATE -fno-exceptions -fno-rtti)
target_link_options(pocketrun_adb PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(pocketrun_adb PRIVATE log boringssl::crypto_static cxx::cxx)
